-- =========================
-- CASE OPERATOR IN SQL
-- =========================

-- Show Databases
SHOW DATABASES;

-- Use Database
USE VITBHOPAL;

-- Display Complete Worker Table
SELECT * FROM Worker;

-- =========================================
-- CASE OPERATOR BASED ON SALARY
-- =========================================

SELECT
    WORKER_ID,
    FIRST_NAME,
    LAST_NAME,
    DEPARTMENT,
    SALARY,

    CASE
        WHEN SALARY >= 400000 THEN 'Very High Salary'
        WHEN SALARY BETWEEN 200000 AND 399999 THEN 'High Salary'
        WHEN SALARY BETWEEN 100000 AND 199999 THEN 'Medium Salary'
        ELSE 'Low Salary'
    END AS Salary_Status

FROM Worker;

-- =========================================
-- CASE OPERATOR BASED ON DEPARTMENT
-- =========================================

SELECT
    WORKER_ID,
    FIRST_NAME,
    DEPARTMENT,

    CASE
        WHEN DEPARTMENT = 'HR' THEN 'Human Resource Department'
        WHEN DEPARTMENT = 'Admin' THEN 'Administration Department'
        WHEN DEPARTMENT = 'Account' THEN 'Accounts Department'
        ELSE 'Other Department'
    END AS Department_Status

FROM Worker;

-- =========================================
-- CASE OPERATOR WITH BONUS
-- =========================================

SELECT
    WORKER_ID,
    FIRST_NAME,
    SALARY,

    CASE
        WHEN SALARY >= 400000 THEN SALARY + 50000
        WHEN SALARY BETWEEN 200000 AND 399999 THEN SALARY + 30000
        ELSE SALARY + 10000
    END AS Updated_Salary

FROM Worker;

-- =========================================
-- CASE OPERATOR WITH PERFORMANCE STATUS
-- =========================================

SELECT
    FIRST_NAME,
    SALARY,

    CASE
        WHEN SALARY >= 500000 THEN 'Excellent Performance'
        WHEN SALARY >= 200000 THEN 'Good Performance'
        ELSE 'Average Performance'
    END AS Performance_Status

FROM Worker;
